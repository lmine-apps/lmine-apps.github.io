// ------------------------------------------------------------
function typeToDisplay_(t) {
  if (!t) return '';
  return TYPE_DISPLAY_NAMES[t] || t;
}
function displayToType_(d) {
  if (!d) return null;
  if (VALID_TYPES.indexOf(d) >= 0) return d;
  for (var k in TYPE_DISPLAY_NAMES) {
    if (TYPE_DISPLAY_NAMES[k] === d) return k;
  }
  return null;
}
function serviceLabel_(key) {
  return SERVICE_LABELS[key] || String(key || '');
}

function doGet(e)  { return handleRequest_(e); }
function doPost(e) { return handleRequest_(e); }

function handleRequest_(e) {
  try {
    var p = mergeParams_(e);
    var action = String(p.action || '').trim();
    if (!action) return json_({ ok: false, error: 'no action' });

    // admin_* は ADMIN_TOKEN 認証
    if (action.indexOf('admin_') === 0) {
      var adminToken = getProp_('ADMIN_TOKEN');
      if (!adminToken || String(p.admin_token) !== adminToken) {
        return json_({ ok: false, error: 'invalid admin token' });
      }
      switch (action) {
        case 'admin_stats':            return handleAdminStats_(p);
        case 'admin_list_users':       return handleAdminListUsers_(p);
        case 'admin_get_user':         return handleAdminGetUser_(p);
        case 'admin_list_interests':   return handleAdminListInterests_(p);
        case 'admin_update_interest':  return handleAdminUpdateInterest_(p);
        case 'admin_register_token':   return handleAdminRegisterToken_(p);
        case 'admin_test_push':        return handleAdminTestPush_(p);
        case 'admin_list_viewers':     return handleAdminListViewers_(p);      // v3.7
        case 'admin_video_stats':      return handleAdminVideoStats_(p);       // v3.7
        case 'admin_post_notice':      return handleAdminPostNotice_(p);       // v3.7
        case 'admin_update_viewer':    return handleAdminUpdateViewer_(p);     // v3.7
        case 'admin_list_notices':     return handleAdminListNotices_(p);      // v3.7
        case 'admin_delete_notice':    return handleAdminDeleteNotice_(p);     // v3.7
        default: return json_({ ok: false, error: 'unknown admin action: ' + action });
      }
    }

    if (String(p.token) !== TOKEN) {
      return json_({ ok: false, error: 'invalid token' });
    }
    switch (action) {
      case 'ping':         return json_({ ok: true, time: fmtDateTime_(new Date()) });
      case 'get_type':     return handleGetType_(p);
      case 'record_event': return handleRecordEvent_(p);
      case 'list_videos':  return handleListVideosV38_(p);   // ★v3.8 (新シート優先)
      case 'check_viewer':       return handleCheckViewer_(p);         // v3.7
      case 'register_viewer':    return handleRegisterViewer_(p);      // v3.7
      case 'update_viewer_name': return handleUpdateViewerName_(p);    // v3.7
      case 'update_viewer':      return handleUpdateViewer_(p);        // v3.7 (email/notify)
      case 'log_view':           return handleLogView_(p);             // v3.7
      case 'toggle_like':        return handleToggleLike_(p);          // v3.7
      case 'get_stats':          return handleGetStats_(p);            // v3.7
      case 'get_user_likes':     return handleGetUserLikes_(p);        // v3.7
      case 'get_notices':        return handleGetNotices_(p);          // v3.7
      default:             return json_({ ok: false, error: 'unknown action: ' + action });
    }
  } catch (err) {
    return json_({ ok: false, error: String(err && err.message || err) });
  }
}

function mergeParams_(e) {
  var out = {};
  if (e && e.parameter) {
    for (var k in e.parameter) out[k] = e.parameter[k];
  }
  if (e && e.postData && e.postData.contents) {
    try {
      var body = JSON.parse(e.postData.contents);
      for (var k2 in body) out[k2] = body[k2];
    } catch (_e) {}
  }
  return out;
}

function json_(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function getProp_(key) {
  return PropertiesService.getScriptProperties().getProperty(key) || '';
}

function handleGetType_(p) {
  var uid = String(p.uid || '').trim();
  if (!uid) return json_({ ok: false, error: 'no uid' });

  var sheet = SpreadsheetApp.getActive().getSheetByName(SHEET_USERS);
  if (!sheet) return json_({ ok: false, error: 'no users sheet' });

  var row = findUserRow_(sheet, uid);
  if (!row) return json_({ ok: true, type: null });

  var stored = sheet.getRange(row, COL.type).getValue();
  var type = displayToType_(stored);
  return json_({ ok: true, type: type });
}

function handleRecordEvent_(p) {
  var uid = String(p.uid || '').trim();
  var type = String(p.type || '').trim();
  var name = String(p.name || '').trim();
  var event = String(p.event || '').trim();
  var payload = p.payload || {};

  if (!uid) return json_({ ok: false, error: 'no uid' });
  if (VALID_EVENTS.indexOf(event) < 0) {
    return json_({ ok: false, error: 'invalid event: ' + event });
  }
  if (type && VALID_TYPES.indexOf(type) < 0) {
    return json_({ ok: false, error: 'invalid type: ' + type });
  }

  var ss = SpreadsheetApp.getActive();
  var now = new Date();
  var nowStr = fmtDateTime_(now);

  var evSheet = ss.getSheetByName(SHEET_EVENTS);
  if (evSheet) {
    var payloadForLog = payload;
    if (name && !payloadForLog.name) payloadForLog = Object.assign({}, payload, { name: name });
    evSheet.appendRow([nowStr, uid, typeToDisplay_(type), event, JSON.stringify(payloadForLog)]);
  }

  var usersSheet = ss.getSheetByName(SHEET_USERS);
  if (!usersSheet) return json_({ ok: false, error: 'no users sheet' });

  var row = findUserRow_(usersSheet, uid);
  if (!row) {
    row = usersSheet.getLastRow() + 1;
    var newRow = [];
    for (var i = 0; i < TOTAL_COLS; i++) newRow.push('');
    newRow[COL.uid - 1] = uid;
    newRow[COL.type - 1] = typeToDisplay_(type);
    if (name) newRow[COL.name - 1] = name;
    newRow[COL.last_event_at - 1] = nowStr;
    usersSheet.getRange(row, 1, 1, TOTAL_COLS).setValues([newRow]);
  }

  var typeCell = usersSheet.getRange(row, COL.type);
  if (!typeCell.getValue() && type) typeCell.setValue(typeToDisplay_(type));

  if (name) {
    var nameCell = usersSheet.getRange(row, COL.name);
    if (!nameCell.getValue()) nameCell.setValue(name);
  }

  var colMap = {
    diagnosis_complete: COL.diagnosis_at,
    app_started:        COL.app_started_at,
    day1_completed:     COL.day1_at,
    day2_completed:     COL.day2_at,
    day3_completed:     COL.day3_at,
    card_created:       COL.day3_at,
    day4_completed:     COL.day4_at,
    report_generated:   COL.day4_at,
    report_viewed:      COL.report_viewed_at,
    offer_viewed:       COL.offer_viewed_at,
    offer_clicked:      COL.offer_clicked_at,
    app_expired:        COL.expired_at
  };
  var col = colMap[event];
  var alreadyRecorded = false;
  if (col) {
    var cell = usersSheet.getRange(row, col);
    if (!cell.getValue()) cell.setValue(nowStr);
    else alreadyRecorded = true;
  }

  if (event === 'app_started') {
    var expCell = usersSheet.getRange(row, COL.expired_at);
    if (!expCell.getValue()) {
      var expMs = now.getTime() + APP_AVAILABLE_HOURS * 3600 * 1000;
      expCell.setValue(fmtDateTime_(new Date(expMs)));
    }
  }

  updateUserStatus_(usersSheet, row);
  usersSheet.getRange(row, COL.last_event_at).setValue(nowStr);

  var beaconResult = null;
  if (event === 'app_started' && !alreadyRecorded) {
    var appStartCell = usersSheet.getRange(row, COL.app_start_notified_at);
    if (!appStartCell.getValue()) {
      beaconResult = fireBeacon_(uid, BEACON_STEPS.app_start_welcome, 'app_started');
      appStartCell.setValue(nowStr);
    }
  }

  // ★v3.5：関心表明イベント → 関心リストシート追記 & FCM
  var interestResult = null;
  if (event === 'interest_expressed') {
    interestResult = recordInterest_(uid, name, type, payload, nowStr);
  }

  return json_({
    ok: true,
    event: event,
    row: row,
    beacon: beaconResult,
    interest: interestResult,
    duplicate: alreadyRecorded
  });
}

function recordInterest_(uid, name, type, payload, nowStr) {
  var ss = SpreadsheetApp.getActive();
  var sheet = ss.getSheetByName(SHEET_INTERESTS);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_INTERESTS);
    sheet.getRange(1, 1, 1, INTEREST_TOTAL_COLS).setValues([[
      '表明日時', 'UID', '名前', 'タイプ', 'サービスキー', 'サービス名',
      'コピーメッセージ', 'ステータス', 'メモ', '通知送信'
    ]]);
  }
  var serviceKey = String(payload.service || '').trim();
  var message = String(payload.message || '').trim();
  var newRow = [];
  for (var i = 0; i < INTEREST_TOTAL_COLS; i++) newRow.push('');
  newRow[ICOL.expressed_at - 1]  = nowStr;
  newRow[ICOL.uid - 1]           = uid;
  newRow[ICOL.name - 1]          = name || '';
  newRow[ICOL.type_display - 1]  = typeToDisplay_(type);
  newRow[ICOL.service_key - 1]   = serviceKey;
  newRow[ICOL.service_label - 1] = serviceLabel_(serviceKey);
  newRow[ICOL.message - 1]       = message;
  newRow[ICOL.status - 1]        = '新規';
  var rowNum = sheet.getLastRow() + 1;
  sheet.getRange(rowNum, 1, 1, INTEREST_TOTAL_COLS).setValues([newRow]);

  var pushResult = null;
  try {
    var title = '💬 関心表明が入りました♡';
    var body = serviceLabel_(serviceKey) + '\n'
             + '（LINEに「' + (message || svcNameFromKey_(serviceKey) + 'を詳しく知りたい！') + '」が届くはず）';
    pushResult = sendFCMToAdmin_(title, body, {
      kind: 'interest',
      uid: uid,
      service: serviceKey
    });
    if (pushResult && pushResult.ok) {
      sheet.getRange(rowNum, ICOL.notified_at).setValue(fmtDateTime_(new Date()));
    }
  } catch (e) {
    pushResult = { ok: false, error: String(e && e.message || e) };
  }
  return { row: rowNum, push: pushResult };
}

function handleListVideos_(p) {
  var cache = CacheService.getScriptCache();
  var cached = cache.get('videos_json_v2');
  if (cached) return ContentService.createTextOutput(cached).setMimeType(ContentService.MimeType.JSON);

  var ss = SpreadsheetApp.openById(VIDEO_SHEET_ID);
  var sheet = ss.getSheetByName(VIDEO_SHEET_NAME);
  if (!sheet) return json_({ ok: false, error: 'no video sheet' });

  var last = sheet.getLastRow();
  if (last < 4) return json_({ ok: true, videos: [] });

  // B〜G列を読む（データは4行目〜）
  var range = sheet.getRange(4, 2, last - 3, 6);
  var values = range.getValues();
  var richs  = range.getRichTextValues();
  var months = range.getDisplayValues();  // 実施日の表示形式

  var videos = [];
  var lastMonth = '';    // セル結合対策：直前の月を継承
  var lastRecipe = '';
  for (var i = 0; i < values.length; i++) {
    var v = values[i];
    var d = months[i];
    var r = richs[i];
    if (!v[1]) continue;  // 番号なしはスキップ

    // 月：空なら直前の月を継承（B列セル結合対策）
    var month = String(d[0] || '').trim();
    if (month) lastMonth = month;
    else month = lastMonth;

    // タイトル：D列が空ならE列(linkText)から抽出
    var title = String(v[2] || '').replace(/\r/g, '').trim();
    if (!title) {
      var linkText = String(v[3] || '');
      // 「④手毬寿司の成型、瓦そば | Vimeo の動画と映画」→「手毬寿司の成型、瓦そば」
      var m = linkText.match(/^[\s]*[⓪①②③④⑤⑥⑦⑧⑨⑩⑪⑫⑬⑭⑮⑯⑰⑱⑲⑳○◯⚫]*[\s]*(.*?)[\s]*[|｜]\s*Vimeo/);
      if (m) title = m[1].trim();
      if (!title) title = linkText.replace(/\s*[|｜]\s*Vimeo.*$/, '').replace(/^[⓪-⓯①-⑳]/, '').trim();
    }

    // レシピ名：空なら直前を継承（G列セル結合対策）
    var recipe = String(v[5] || '').trim();
    if (recipe) lastRecipe = recipe;
    else if (extractUrl_(r[5])) recipe = ''; // 個別リンクあり→独立
    else recipe = lastRecipe;

    var vimeoUrl = extractUrl_(r[3]);
    var recipeUrl = extractUrl_(r[5]) || (lastRecipe && extractUrl_(richs[Math.max(0, i-1)][5])) || '';

    // 空行スキップ：タイトル・linkText・Vimeoリンク全部空なら未実施行
    if (!title && !v[3] && !vimeoUrl) continue;

    videos.push({
      month:     month,
      no:        Number(v[1]),
      title:     title,
      linkText:  String(v[3] || ''),
      duration:  String(d[4] || ''),
      recipe:    recipe,
      vimeoUrl:  vimeoUrl,
      recipeUrl: recipeUrl,
      series:    detectSeries_(title)
    });
  }

  var out = JSON.stringify({ ok: true, videos: videos, count: videos.length, cached_at: fmtDateTime_(new Date()) });
  try { cache.put('videos_json_v2', out, VIDEO_CACHE_SEC); } catch (e) {}
  return ContentService.createTextOutput(out).setMimeType(ContentService.MimeType.JSON);
}

function extractUrl_(richText) {
  if (!richText) return '';
  var url = richText.getLinkUrl();
  if (url) return url;
  // Runs 内も探す
  var runs = richText.getRuns();
  for (var i = 0; i < runs.length; i++) {
    var u = runs[i].getLinkUrl();
    if (u) return u;
  }
  return '';
}

function detectSeries_(title) {
  if (/家で料亭|料亭の味/.test(title))   return '家で料亭の味';
  if (/自分史上最高/.test(title))         return '自分史上最高';
  if (/べきべき/.test(title))             return 'べきべき卒業';
  if (/発酵|麹|塩麹|醤油麹/.test(title))  return '発酵・麹';
  if (/ラブキチ/.test(title))             return 'ラブキチ';
  return '';
}

function svcNameFromKey_(key) {
  var label = SERVICE_LABELS[key] || '';
  // ラベルの「（年間）」等を取り除いた短い名前を返す
  return label.replace(/（.*?）/g, '');
}

function morningNotifications() {
  var now = new Date();
  var hour = parseInt(Utilities.formatDate(now, 'Asia/Tokyo', 'HH'), 10);
  if (TEST_MODE) {
    processMorning_(now);
    processEvening_(now);
    return;
  }
  if (hour === MORNING_HOUR)      processMorning_(now);
  else if (hour === EVENING_HOUR) processEvening_(now);
  else Logger.log('morningNotifications: skip (hour=' + hour + ')');
}

function processMorning_(now) {
  var sheet = SpreadsheetApp.getActive().getSheetByName(SHEET_USERS);
  if (!sheet) return;
  var last = sheet.getLastRow();
  if (last < 2) return;

  var vals = sheet.getRange(2, 1, last - 1, TOTAL_COLS).getValues();
  var today = fmtDate_(now);
  var stats = { checked: 0, morning: 0, day4: 0, expired: 0, skipped: 0 };

  for (var i = 0; i < vals.length; i++) {
    stats.checked++;
    var row = i + 2;
    var uid = String(vals[i][COL.uid - 1] || '');
    if (!uid) continue;

    var appStarted   = vals[i][COL.app_started_at - 1];
    var day3At       = vals[i][COL.day3_at - 1];
    var day4At       = vals[i][COL.day4_at - 1];
    var expiredAt    = vals[i][COL.expired_at - 1];
    var lastMorning  = vals[i][COL.last_morning_notified_date - 1];
    var day4Ready    = vals[i][COL.day4_ready_notified_at - 1];
    var expReminder  = vals[i][COL.expired_reminder_notified_at - 1];

    if (!appStarted) continue;
    var expDate = parseDateSafe_(expiredAt);
    if (expDate && expDate.getTime() < now.getTime()) continue;

    if (day3At && !day4At && !day4Ready) {
      fireBeacon_(uid, BEACON_STEPS.day4_ready, 'day4_ready_check');
      sheet.getRange(row, COL.day4_ready_notified_at).setValue(fmtDateTime_(now));
      stats.day4++;
      continue;
    }

    if (expDate && !expReminder) {
      var hoursLeft = (expDate.getTime() - now.getTime()) / 3600000;
      if (hoursLeft <= EXPIRED_REMINDER_HOURS && hoursLeft > 0) {
        fireBeacon_(uid, BEACON_STEPS.expired_soon, 'expired_soon_check');
        sheet.getRange(row, COL.expired_reminder_notified_at).setValue(fmtDateTime_(now));
        stats.expired++;
        continue;
      }
    }

    if (day4At) { stats.skipped++; continue; }
    if (String(lastMorning) === today) { stats.skipped++; continue; }

    fireBeacon_(uid, BEACON_STEPS.morning_daily, 'morning_daily');
    sheet.getRange(row, COL.last_morning_notified_date).setValue(today);
    stats.morning++;
  }
  Logger.log('processMorning: ' + JSON.stringify(stats));
}

function processEvening_(now) {
  var sheet = SpreadsheetApp.getActive().getSheetByName(SHEET_USERS);
  if (!sheet) return;
  var last = sheet.getLastRow();
  if (last < 2) return;

  var vals = sheet.getRange(2, 1, last - 1, TOTAL_COLS).getValues();
  var today = fmtDate_(now);
  var stats = { checked: 0, reminded: 0, skipped: 0 };

  for (var i = 0; i < vals.length; i++) {
    stats.checked++;
    var row = i + 2;
    var uid = String(vals[i][COL.uid - 1] || '');
    if (!uid) continue;

    var appStarted = vals[i][COL.app_started_at - 1];
    var day4At     = vals[i][COL.day4_at - 1];
    var expiredAt  = vals[i][COL.expired_at - 1];
    var lastEvening = vals[i][COL.last_evening_reminded_date - 1];

    if (!appStarted) continue;
    if (day4At) { stats.skipped++; continue; }
    var expDate = parseDateSafe_(expiredAt);
    if (expDate && expDate.getTime() < now.getTime()) { stats.skipped++; continue; }
    if (String(lastEvening) === today) { stats.skipped++; continue; }

    var completedToday = false;
    for (var dN = 1; dN <= 4; dN++) {
      var colKey = 'day' + dN + '_at';
      var dAt = vals[i][COL[colKey] - 1];
      if (dAt) {
        var d = parseDateSafe_(dAt);
        if (d && fmtDate_(d) === today) { completedToday = true; break; }
      }
    }
    if (completedToday) { stats.skipped++; continue; }

    fireBeacon_(uid, BEACON_STEPS.evening_reminder, 'evening_reminder');
    sheet.getRange(row, COL.last_evening_reminded_date).setValue(today);
    stats.reminded++;
  }
  Logger.log('processEvening: ' + JSON.stringify(stats));
}

function fmtDateTime_(date) {
  return Utilities.formatDate(date, 'Asia/Tokyo', 'yyyy/MM/dd HH:mm');
}
function fmtDate_(date) {
  return Utilities.formatDate(date, 'Asia/Tokyo', 'yyyy-MM-dd');
}
function parseDateSafe_(s) {
  if (!s) return null;
  if (s instanceof Date) return isNaN(s.getTime()) ? null : s;
  var str = String(s);
  var m = str.match(/^(\d{4})\/(\d{1,2})\/(\d{1,2})\s+(\d{1,2}):(\d{1,2})/);
  if (m) {
    var d = new Date(Date.UTC(+m[1], +m[2]-1, +m[3], +m[4]-9, +m[5]));
    return isNaN(d.getTime()) ? null : d;
  }
  var d2 = new Date(str);
  return isNaN(d2.getTime()) ? null : d2;
}

function fireBeacon_(uid, stepId, triggerEvent) {
  if (!stepId) {
    logBeacon_(uid, stepId || '(未設定)', triggerEvent, 0, 'skip');
    return { fired: false, reason: 'no step_id' };
  }
  var url = 'https://autosns.jp/api/call-beacon/' + encodeURIComponent(stepId) + '/' + encodeURIComponent(uid);
  var status = 0;
  var result = 'error';
  try {
    var res = UrlFetchApp.fetch(url, { muteHttpExceptions: true });
    status = res.getResponseCode();
    result = (status >= 200 && status < 300) ? 'success' : 'error';
  } catch (e) {
    result = 'exception: ' + String(e && e.message || e);
  }
  logBeacon_(uid, stepId, triggerEvent, status, result);
  return { fired: true, status: status, result: result };
}

function logBeacon_(uid, stepId, triggerEvent, status, result) {
  var sheet = SpreadsheetApp.getActive().getSheetByName(SHEET_BEACONS);

function sendFCMToAdmin_(title, body, data) {
  var token = getProp_('ADMIN_FCM_TOKEN');
  if (!token) return { ok: false, error: 'ADMIN_FCM_TOKEN not registered' };
  return sendFCM_(token, title, body, data || {});

function sendFCM_(fcmToken, title, body, data) {
  var projectId = getProp_('FCM_PROJECT_ID');
  if (!projectId) return { ok: false, error: 'FCM_PROJECT_ID not set' };
  var accessToken = getFCMAccessToken_();
  if (!accessToken) return { ok: false, error: 'no access token' };

  var url = 'https://fcm.googleapis.com/v1/projects/' + projectId + '/messages:send';
  var strData = {};
  for (var k in (data || {})) strData[k] = String(data[k]);

  var message = {
    message: {
      token: fcmToken,
      notification: { title: title, body: body },
      data: strData,
      webpush: {
        notification: {
          title: title,
          body: body,
          icon: 'https://apps.l-mine.com/kayomama-admin/icons/icon-192.png',
          badge: 'https://apps.l-mine.com/kayomama-admin/icons/badge.png',
          click_action: 'https://apps.l-mine.com/kayomama-admin/'
        },
        fcm_options: { link: 'https://apps.l-mine.com/kayomama-admin/' }
      }
    }
  };
  var res = UrlFetchApp.fetch(url, {
    method: 'post',
    contentType: 'application/json',
    headers: { Authorization: 'Bearer ' + accessToken },
    payload: JSON.stringify(message),
    muteHttpExceptions: true
  });
  var code = res.getResponseCode();
  var txt = res.getContentText();
  return { ok: (code >= 200 && code < 300), status: code, body: txt };
}

function getFCMAccessToken_() {
  var cached = PropertiesService.getScriptProperties().getProperty('FCM_ACCESS_TOKEN_CACHE');
  var exp = parseInt(getProp_('FCM_ACCESS_TOKEN_EXP') || '0', 10);
  if (cached && exp && exp > (Date.now() / 1000) + 60) return cached;

  var privateKey = getProp_('FCM_PRIVATE_KEY');
  var clientEmail = getProp_('FCM_CLIENT_EMAIL');
  if (!privateKey || !clientEmail) return null;

  var iat = Math.floor(Date.now() / 1000);
  var exp2 = iat + 3600;
  var header = { alg: 'RS256', typ: 'JWT' };
  var claim = {
    iss: clientEmail,
    scope: 'https://www.googleapis.com/auth/firebase.messaging',
    aud: 'https://oauth2.googleapis.com/token',
    exp: exp2,
    iat: iat
  };
  var toSign = base64url_(JSON.stringify(header)) + '.' + base64url_(JSON.stringify(claim));
  var sig = Utilities.computeRsaSha256Signature(toSign, privateKey);
  var jwt = toSign + '.' + base64url_(sig);

  var res = UrlFetchApp.fetch('https://oauth2.googleapis.com/token', {
    method: 'post',
    payload: {
      grant_type: 'urn:ietf:params:oauth:grant-type:jwt-bearer',
      assertion: jwt
    },
    muteHttpExceptions: true
  });
  var code = res.getResponseCode();
  if (code < 200 || code >= 300) {
    Logger.log('OAuth error: ' + res.getContentText());
    return null;
  }
  var obj = JSON.parse(res.getContentText());
  var accessToken = obj.access_token;
  PropertiesService.getScriptProperties().setProperties({
    FCM_ACCESS_TOKEN_CACHE: accessToken,
    FCM_ACCESS_TOKEN_EXP: String(iat + (obj.expires_in || 3600) - 60)
  });
  return accessToken;
}

function base64url_(input) {
  var bytes;
  if (typeof input === 'string') bytes = Utilities.newBlob(input).getBytes();
  else bytes = input;
  var s = Utilities.base64EncodeWebSafe(bytes);
  return s.replace(/=+$/, '');
}

function handleGetStats_(p) {
  var cache = CacheService.getScriptCache();
  var cached = cache.get('stats_json');
  if (cached) return ContentService.createTextOutput(cached).setMimeType(ContentService.MimeType.JSON);
  var ss = getVideoSs_();
  var log = ensureSheet_(ss, VIEW_LOG_SHEET, VIEW_LOG_COLS);
  var data = log.getDataRange().getValues();
  var stats = {}; var userLikes = {};
  for (var i = 1; i < data.length; i++) {
    var no = Number(data[i][2]), uid = String(data[i][1]), ev = String(data[i][3]);
    if (!stats[no]) stats[no] = { views:0, likes:0 };
    if (ev === 'view') stats[no].views++;
    if (ev === 'like') { if(!userLikes[no])userLikes[no]={}; userLikes[no][uid]=true; }
    if (ev === 'unlike') { if(userLikes[no])delete userLikes[no][uid]; }
  }
  for (var n in userLikes) {
    if (!stats[n]) stats[n] = { views:0, likes:0 };
    stats[n].likes = Object.keys(userLikes[n]).length;
  }
  var out = JSON.stringify({ ok:true, stats:stats });
  try { cache.put('stats_json', out, LIKES_CACHE_SEC); } catch(e){}
  return ContentService.createTextOutput(out).setMimeType(ContentService.MimeType.JSON);
}

function handleGetUserLikes_(p) {
  var uid = String(p.uid||'').trim();

}

var VIDEOS_NEW_SHEET = 'videos';
var VIDEOS_NEW_HEADERS = ['No','種別','カテゴリ','実施日','タイトル','Vimeo URL','動画長','レシピURL','閲覧タグ','シリーズ','タイムテーブル','説明','公開状態'];

var KIND_OPTIONS = ['月次レッスン','単発','有料コース','ライブアーカイブ','特別'];
var CATEGORY_OPTIONS = ['サブスク','有料単発','無料','インスタライブ','限定コース'];
var STATUS_OPTIONS = ['公開','非公開','下書き'];

// migrate_videos_to_new_sheet(): シート1から videos シートへ整形コピー
function migrate_videos_to_new_sheet() {
  var ss = getVideoSs_();
  var oldSheet = ss.getSheetByName(VIDEO_SHEET_NAME);
  if (!oldSheet) throw new Error('シート1 not found');

  // 新シート作成 or リセット
  var newSheet = ss.getSheetByName(VIDEOS_NEW_SHEET);
  if (newSheet) ss.deleteSheet(newSheet);
  newSheet = ss.insertSheet(VIDEOS_NEW_SHEET);

  // ヘッダー行
  newSheet.getRange(1, 1, 1, VIDEOS_NEW_HEADERS.length).setValues([VIDEOS_NEW_HEADERS])
    .setFontWeight('bold').setBackground('#fdeaef').setHorizontalAlignment('center');
  newSheet.setFrozenRows(1);

  // 列幅設定
  var widths = [50, 100, 100, 110, 220, 90, 70, 90, 130, 140, 160, 180, 80];
  for (var w = 0; w < widths.length; w++) newSheet.setColumnWidth(w+1, widths[w]);

  // 旧データ読み込み（handleListVideos_ と同じ整形ロジックを使う）
  var last = oldSheet.getLastRow();
  var range = oldSheet.getRange(4, 2, last - 3, 6);
  var values = range.getValues();
  var richs  = range.getRichTextValues();
  var displays = range.getDisplayValues();

  var rows = [];
  var lastMonth = '';
  var lastRecipeUrl = '';
  for (var i = 0; i < values.length; i++) {
    var v = values[i];
    var d = displays[i];
    var r = richs[i];
    if (!v[1]) continue;

    var month = String(d[0] || '').trim();
    if (month) lastMonth = month; else month = lastMonth;

    var title = String(v[2] || '').replace(/\r/g, '').trim();
    if (!title) {
      var lt = String(v[3] || '');
      var m = lt.match(/^[\s]*[⓪①②③④⑤⑥⑦⑧⑨⑩⑪⑫⑬⑭⑮⑯⑰⑱⑲⑳○◯⚫]*[\s]*(.*?)[\s]*[|｜]\s*Vimeo/);
      if (m) title = m[1].trim();
      if (!title) title = lt.replace(/\s*[|｜]\s*Vimeo.*$/, '').trim();
    }

    var vimeoUrl = extractUrl_(r[3]);
    var recipeUrl = extractUrl_(r[5]);
    if (recipeUrl) lastRecipeUrl = recipeUrl;

    var origLink = String(v[3] || '').trim();
    var origRecipe = String(v[5] || '').trim();

    // 完全空行はスキップ
    if (!title && !origLink && !vimeoUrl && !origRecipe) continue;

    // タイトル空でも一旦linkTextを入れる
    if (!title) title = origLink || '(タイトル未設定)';

    rows.push([
      Number(v[1]),                 // No
      '月次レッスン',                 // 種別
      'サブスク',                    // カテゴリ
      month,                        // 実施日
      title,                        // タイトル
      vimeoUrl,                     // Vimeo URL
      String(d[4] || '').trim(),    // 動画長
      recipeUrl,                    // レシピURL
      'サブスク',                    // 閲覧タグ
      detectSeries_(title),         // シリーズ
      '',                           // タイムテーブル（未入力）
      '',                           // 説明（未入力）
      '公開'                        // 公開状態
    ]);
  }

  if (rows.length > 0) {
    newSheet.getRange(2, 1, rows.length, VIDEOS_NEW_HEADERS.length).setValues(rows);
  }

  // データ入力規則（プルダウン）
  var addValidation_ = function(col, options) {
    var rule = SpreadsheetApp.newDataValidation().requireValueInList(options, true).setAllowInvalid(true).build();
    newSheet.getRange(2, col, 1000, 1).setDataValidation(rule);
  };
  addValidation_(2, KIND_OPTIONS);      // B: 種別
  addValidation_(3, CATEGORY_OPTIONS);  // C: カテゴリ
  addValidation_(13, STATUS_OPTIONS);   // M: 公開状態

  // 折り返し
  newSheet.getRange(2, 5, rows.length, 1).setWrap(true);   // タイトル
  newSheet.getRange(2, 11, rows.length, 1).setWrap(true);  // タイムテーブル
  newSheet.getRange(2, 12, rows.length, 1).setWrap(true);  // 説明

  // キャッシュ無効化
  try { CacheService.getScriptCache().remove('videos_json_v2'); } catch(e) {}
  try { CacheService.getScriptCache().remove('videos_json_v3'); } catch(e) {}

  Logger.log('migrated: ' + rows.length + ' videos to sheet "' + VIDEOS_NEW_SHEET + '"');
  return { ok: true, count: rows.length };
}

// v3.8: 新シートを優先的に読む、なければ旧シート
function handleListVideosV38_(p) {
  var cache = CacheService.getScriptCache();
  var cached = cache.get('videos_json_v3');
  if (cached) return ContentService.createTextOutput(cached).setMimeType(ContentService.MimeType.JSON);

  var ss = getVideoSs_();
  var newSheet = ss.getSheetByName(VIDEOS_NEW_SHEET);

  // 新シートがまだ無い場合は従来のロジックで返す
  if (!newSheet) return handleListVideos_(p);

  var last = newSheet.getLastRow();
  if (last < 2) return json_({ ok: true, videos: [] });

  var range = newSheet.getRange(2, 1, last - 1, VIDEOS_NEW_HEADERS.length);
  var values = range.getValues();
  var richs  = range.getRichTextValues();

  var videos = [];
  for (var i = 0; i < values.length; i++) {
    var v = values[i];
    if (!v[0] && !v[4]) continue;      // No もタイトルも無ければskip
    var status = String(v[12] || '公開').trim();
    if (status && status !== '公開') continue;   // 非公開/下書きは返さない

    // URL列は生値がURLになってればそのまま。ハイパーリンク形式ならextractUrl_
    var vimeoUrl = String(v[5] || '').trim();
    if (!vimeoUrl) vimeoUrl = extractUrl_(richs[i][5]) || '';
    var recipeUrl = String(v[7] || '').trim();
    if (!recipeUrl) recipeUrl = extractUrl_(richs[i][7]) || '';

    var tags = String(v[8] || '').split(/[,、]/).map(function(s){return s.trim();}).filter(Boolean);

    videos.push({
      no:        Number(v[0]),
      kind:      String(v[1] || '').trim(),
      category:  String(v[2] || '').trim(),
      month:     String(v[3] || '').trim(),
      title:     String(v[4] || '').replace(/\r/g, ''),
      vimeoUrl:  vimeoUrl,
      duration:  String(v[6] || '').trim(),
      recipeUrl: recipeUrl,
      tags:      tags,
      series:    String(v[9] || '').trim(),
      timetable: String(v[10] || '').trim(),
      description: String(v[11] || '').trim(),
      linkText:  '',   // 互換
      recipe:    ''    // 互換
    });
  }

  var out = JSON.stringify({ ok:true, videos:videos, count:videos.length, cached_at:fmtDateTime_(new Date()) });
  try { cache.put('videos_json_v3', out, VIDEO_CACHE_SEC); } catch(e) {}
  return ContentService.createTextOutput(out).setMimeType(ContentService.MimeType.JSON);
}
